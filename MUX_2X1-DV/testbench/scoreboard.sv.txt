class scoreboard;
  transaction ref_pkt, dut_pkt;
  mailbox im2s;
  mailbox om2s;

  function new(mailbox im2s, mailbox om2s);
    this.im2s = im2s;
    this.om2s = om2s;
  endfunction

  task run();
    repeat (4) begin
      im2s.get(ref_pkt);
      om2s.get(dut_pkt);

      $display("\n===== SCOREBOARD =====");
      $display("Expected output = %0d", ref_pkt.out);
      $display("Actual output   = %0d", dut_pkt.out);

      if (ref_pkt.out === dut_pkt.out) begin
        $display("TEST PASSED at time %0t", $time);
      end
      else begin
        $error("TEST FAILED at time %0t", $time);
      end
    end
  endtask
endclass
