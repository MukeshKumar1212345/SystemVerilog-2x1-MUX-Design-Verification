class output_monitor;
  transaction pkt;
  mailbox om2s;

  virtual interface mux_2x1_intf.tb inter;

  function new(mailbox im2s,
               virtual interface mux_2x1_intf.tb inter);
    this.om2s = im2s;
    this.inter = inter;
  endfunction

  task run();
    forever begin
      @(inter.out);

      pkt = new();
      pkt.out = inter.out;

      $display("\n>>>>OUTPUT MONITOR<<<<");
      $display("[Op monitor]: stored data in output monitor as packet data = %0d",
               pkt.out);

      om2s.put(pkt);
      $display("[op monitor to mailbox]: data sent from op monitor to mailbox = %0d",
               pkt.out);
    end
  endtask
endclass
